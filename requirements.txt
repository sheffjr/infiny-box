# Runtime dependencies of the Infiny CLI.
#
# This list is what `Dockerfile` installs into /opt/infiny/venv, in the same
# order and with nothing extra. Keep the two in step: they diverged once and
# this file grew three packages the product never imported — fastapi, uvicorn
# and websockets, left over from a web-UI that Phase 1 does not have. `cli.py`
# tells users to run `pip install -r requirements.txt` when rich is missing,
# so a wrong list here is a wrong instruction, not just untidy bookkeeping.
#
# Hermes itself is NOT here. It lives in its own venv (/opt/hermes-venv) with
# its own pinned dependencies — mixing them with ours is a reliable way to get
# a version conflict for free.

httpx>=0.27
psutil>=6.1
pyyaml>=6.0

# The rich terminal client. Without these, cli.py falls back to the
# stdlib-only --plain mode rather than failing.
rich>=13.9
prompt_toolkit>=3.0

# Hides the message text from /proc/PID/cmdline. Without it an agent running
# `pkill -f <word from the request>` kills the CLI that asked it.
setproctitle>=1.3
